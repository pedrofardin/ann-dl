    """Configuração compartilhada pelos exercícios: semente, pastas, dados e figuras."""

from pathlib import Path

import matplotlib

matplotlib.use("Agg")  # gera os PNG sem abrir janela
import matplotlib.pyplot as plt  # noqa: E402
import numpy as np  # noqa: E402
import torch  # noqa: E402
from torch.utils.data import DataLoader, random_split  # noqa: E402
from torchvision import datasets, transforms  # noqa: E402

SEED = 42
CODE_DIR = Path(__file__).resolve().parent
FIGURES = CODE_DIR.parent / "figures"
DATA_DIR = CODE_DIR.parents[3] / "data"  # raiz do repositório; fica fora do Git (.gitignore)

torch.use_deterministic_algorithms(True)  # mesma execução, mesmos números


def _stack(subset):
    """Junta as imagens de um Dataset num tensor (N, 784) e os rótulos num tensor (N,)."""
    xs, ys = next(iter(DataLoader(subset, batch_size=len(subset))))
    return xs.view(len(subset), -1), ys


def load_mnist():
    """Item 1A: MNIST com pixels em [0, 1], achatado em vetores de 784.

    A divisão 50 000 / 10 000 é exatamente a do enunciado. O conjunto de teste volta separado e
    só é usado depois do treino, para os números finais e as figuras.
    """
    train_full = datasets.MNIST(DATA_DIR, train=True, download=True, transform=transforms.ToTensor())
    test = datasets.MNIST(DATA_DIR, train=False, download=True, transform=transforms.ToTensor())
    train, val = random_split(train_full, [50_000, 10_000],
                              generator=torch.Generator().manual_seed(42))
    return {"train": _stack(train), "val": _stack(val), "test": _stack(test)}


def save(fig, name):
    """Salva a figura em figures/ e libera a memória."""
    FIGURES.mkdir(parents=True, exist_ok=True)
    fig.savefig(FIGURES / name, dpi=150, bbox_inches="tight")
    plt.close(fig)


def tile(images, rows, cols):
    """Monta (rows·cols, 784) imagens num mosaico (rows·28, cols·28), na ordem das linhas."""
    imgs = np.asarray(images).reshape(rows, cols, 28, 28)
    return imgs.transpose(0, 2, 1, 3).reshape(rows * 28, cols * 28)


def loss_curves_figure(history, title, name):
    """Dois painéis: reconstrução × época e KL × época, cada um com treino e validação."""
    epochs = np.arange(1, len(history["train_rec"]) + 1)
    fig, axes = plt.subplots(1, 2, figsize=(12, 4.5), layout="constrained")
    for ax, term, label in [(axes[0], "rec", "Reconstrução (BCE)"), (axes[1], "kl", "KL")]:
        ax.plot(epochs, history[f"train_{term}"], "o-", lw=2, ms=4, label="Treino (média da época)")
        ax.plot(epochs, history[f"val_{term}"], "s-", lw=2, ms=4, label="Validação (fim da época)")
        ax.set_xlabel("Época")
        ax.set_ylabel(f"{label} (nats por imagem)")
        ax.set_title(f"{label} × época")
        ax.set_xticks(epochs[::2])
        ax.grid(alpha=0.3)
        ax.legend()
    fig.suptitle(title)
    save(fig, name)

    Abordagem e desafios

    Abordagem. O modelo, a perda e o laço de treino ficam num único arquivo, code/vae.py. O autograd do PyTorch calcula os gradientes, como o enunciado permite. As partes próprias do VAE estão escritas à mão: a reparametrização, a BCE somada nos pixels, a KL em forma fechada e a amostragem do prior. Não uso nenhuma classe de VAE de biblioteca nem torch.distributions. O Exercício 2 chama a mesma função train com d = 2: só esse argumento muda.

    Desafios.

    • Manter as duas parcelas em nats por imagem. F.binary_cross_entropy(..., reduction="sum") soma nos 784 pixels e no lote; depois divido pelo tamanho do lote. A KL também soma nas \(d\) dimensões e divide pelo lote. Com reduction="mean", a BCE ficaria 784 vezes menor e seria o mesmo que usar \(\beta \approx 784\).
    • Não deixar a avaliação mexer no treino. A perda de validação também sorteia \(z\). Se o sorteio usasse o gerador global, cada avaliação mudaria a ordem dos lotes da época seguinte. Por isso, evaluate usa um gerador próprio, com semente fixa: o número de validação é sempre o mesmo e o treino não muda.
    • O teste só no fim. train recebe só o treino e a validação. O teste entra depois que os dois modelos terminam, para os números finais e as Figuras 2 a 6. Nenhuma decisão olha o teste.
    • Medir, e não só olhar. As perguntas de análise pedem números. Por isso, o script mede também a KL de cada dimensão latente, a distância entre as amostras do prior e os códigos dos dados, a fração de pixels cinzentos e a sobreposição das classes no espaço latente.

    """Exercício 1 — Um VAE no MNIST, com d = 16.

Treina o modelo, calcula os números de teste e produz as Figuras 1, 2 e 3. Também mede os
números que sustentam as respostas do item 1E.
"""

import matplotlib.pyplot as plt
import torch

from common import loss_curves_figure, save, tile
from vae import evaluate, kl_per_dim, loss_terms, sample_prior, train

D = 16


def first_of_each_digit(y):
    """Índice da primeira imagem de cada dígito 0–9 no conjunto."""
    return [int((y == k).nonzero()[0]) for k in range(10)]


def figure2(model, x_test, idx):
    """Figura 2: original (em cima) e reconstrução decodificando μ (embaixo)."""
    x = x_test[idx]
    with torch.no_grad():
        mu, _ = model.encode(x)
        x_hat = model.decode(mu)  # decodifica μ: nada de sortear z
    fig, axes = plt.subplots(2, 10, figsize=(12, 2.9), layout="constrained")
    for k in range(10):
        for row, img in enumerate([x[k], x_hat[k]]):
            ax = axes[row, k]
            ax.imshow(img.view(28, 28), cmap="gray", vmin=0, vmax=1)
            ax.set_xticks([])
            ax.set_yticks([])
        axes[0, k].set_title(f"{k} (#{idx[k]})", fontsize=9)
    axes[0, 0].set_ylabel("Original")
    axes[1, 0].set_ylabel("Recon. de μ")
    fig.suptitle(f"Figura 2 — Primeira imagem de teste de cada dígito e sua reconstrução (d = {D})")
    fig.supxlabel("Dígito (índice no conjunto de teste)")
    save(fig, "fig2_reconstructions.png")


def figure3(samples):
    """Figura 3: grade 8 × 8 de imagens decodificadas de z ~ N(0, I)."""
    fig, ax = plt.subplots(figsize=(7, 7.3), layout="constrained")
    ax.imshow(tile(samples, 8, 8), cmap="gray", vmin=0, vmax=1)
    ax.set_xticks([28 * c + 13.5 for c in range(8)], range(1, 9))
    ax.set_yticks([28 * r + 13.5 for r in range(8)], range(1, 9))
    ax.set_xlabel("Coluna da grade")
    ax.set_ylabel("Linha da grade")
    ax.set_title(f"Figura 3 — 64 amostras do prior: x̂ = decoder(z), z ~ N(0, I), d = {D}")
    save(fig, "fig3_samples.png")


def gray_fraction(x):
    """Fração dos pixels com valor em (0.1, 0.9): nem apagados nem acesos."""
    return ((x > 0.1) & (x < 0.9)).float().mean().item()


@torch.no_grad()
def analysis_numbers(model, x_val, x_test, samples):
    """Os números do item 1E."""
    mu, logvar = model.encode(x_test)
    kl_dims = kl_per_dim(mu, logvar)

    # Onde as amostras do prior caem: distância de cada z ~ N(0, I) ao μ de teste mais próximo,
    # comparada com a distância de um μ de validação ao μ de teste mais próximo. Só nas dimensões
    # ativas: nas outras, μ ≈ 0 e o decoder ignora a coordenada, que só somaria ruído do prior.
    active = kl_dims > 0.1
    g = torch.Generator().manual_seed(1)
    z_prior = torch.randn(len(x_test), D, generator=g)
    mu_val, _ = model.encode(x_val)
    nn_prior = torch.cdist(z_prior[:, active], mu[:, active]).min(dim=1).values
    nn_val = torch.cdist(mu_val[:, active], mu[:, active]).min(dim=1).values

    # O que o decoder devolve: probabilidades, não pixels 0/1.
    x_hat_mu = model.decode(mu)
    rec_mu = loss_terms(x_test, x_hat_mu, mu, logvar)[0].item()

    return {
        "kl_per_dim_test": [round(v, 3) for v in kl_dims.tolist()],
        "active_dims_kl_gt_0.1": int(active.sum()),
        "mean_posterior_sigma_test": torch.exp(0.5 * logvar).mean().item(),
        "aggregate_var_per_dim_mean": (mu.var(dim=0) + logvar.exp().mean(dim=0)).mean().item(),
        "nn_dist_prior_to_test_mu_median": nn_prior.median().item(),
        "nn_dist_val_mu_to_test_mu_median": nn_val.median().item(),
        "frac_prior_farther_than_p95_val": (nn_prior > nn_val.quantile(0.95)).float().mean().item(),
        "gray_fraction_test_images": gray_fraction(x_test),
        "gray_fraction_recon_mu": gray_fraction(x_hat_mu),
        "gray_fraction_prior_samples": gray_fraction(samples),
        "test_rec_decoding_mu": rec_mu,
    }


def run(data):
    (x_train, _), (x_val, _), (x_test, y_test) = data["train"], data["val"], data["test"]
    model, history = train(D, x_train, x_val)

    # Daqui para baixo o treino acabou: o teste entra só para os números finais e as figuras.
    test = evaluate(model, x_test)
    loss_curves_figure(history, f"Figura 1 — As duas parcelas da perda, treino e validação (d = {D})",
                       "fig1_loss_terms.png")

    idx = first_of_each_digit(y_test)
    figure2(model, x_test, idx)

    torch.manual_seed(0)  # semente do enunciado para a Figura 3
    samples = sample_prior(model, 64)
    figure3(samples)

    results = {
        "d": D,
        "test": test,
        "history": history,
        "val_kl_first_epoch": history["val_kl"][0],
        "val_kl_last_epoch": history["val_kl"][-1],
        "figure2_test_indices": idx,
        "analysis": analysis_numbers(model, x_val, x_test, samples),
    }
    return model, results

    """O VAE do item 1B: modelo, reparametrização, as duas parcelas da perda, amostragem e treino.

O autograd do PyTorch calcula os gradientes. As partes próprias do VAE estão escritas à mão:
a reparametrização (`reparameterize`), a BCE somada nos pixels e a KL em forma fechada
(`loss_terms`) e a amostragem do prior (`sample_prior`). Não há classe de VAE de biblioteca
nem `torch.distributions`.
"""

import torch
import torch.nn.functional as F
from torch import nn

PIXELS = 784


class VAE(nn.Module):
    """Encoder 784 → 512 → 256 (ReLU), cabeças μ e log σ² 256 → d, decoder d → 256 → 512 → 784."""

    def __init__(self, d):
        super().__init__()
        self.d = d
        self.encoder = nn.Sequential(
            nn.Linear(PIXELS, 512), nn.ReLU(),
            nn.Linear(512, 256), nn.ReLU(),
        )
        self.mu_head = nn.Linear(256, d)
        self.logvar_head = nn.Linear(256, d)  # a rede prevê log σ²: qualquer real vale, σ² > 0 sempre
        self.decoder = nn.Sequential(
            nn.Linear(d, 256), nn.ReLU(),
            nn.Linear(256, 512), nn.ReLU(),
            nn.Linear(512, PIXELS), nn.Sigmoid(),  # cada saída é a probabilidade do pixel estar "aceso"
        )

    def encode(self, x):
        """q(z | x) = N(μ, diag(σ²)): devolve μ e log σ², ambos (N, d)."""
        h = self.encoder(x)
        return self.mu_head(h), self.logvar_head(h)

    def decode(self, z):
        """p(x | z): devolve x̂ (N, 784), a média de uma Bernoulli por pixel."""
        return self.decoder(z)

    def forward(self, x, generator=None):
        mu, logvar = self.encode(x)
        z = reparameterize(mu, logvar, generator)
        return self.decode(z), mu, logvar


def reparameterize(mu, logvar, generator=None):
    """z = μ + σ ⊙ ε, com ε ~ N(0, I).

    O sorteio fica em ε, que não depende dos parâmetros. Assim z é uma função diferenciável
    de μ e de σ, e o gradiente da perda chega ao encoder. σ = exp(½ log σ²).
    """
    eps = torch.randn(mu.shape, generator=generator)
    return mu + torch.exp(0.5 * logvar) * eps


def loss_terms(x, x_hat, mu, logvar):
    """As duas parcelas da perda, em nats por imagem (soma nos pixels e nas dimensões, média no lote).

    Reconstrução: −Σ_j [x_j log x̂_j + (1 − x_j) log(1 − x̂_j)], a BCE somada nos 784 pixels.
    KL(q(z|x) ‖ N(0, I)) em forma fechada: ½ Σ_i (μ_i² + σ_i² − 1 − log σ_i²).
    """
    n = x.shape[0]
    rec = F.binary_cross_entropy(x_hat, x, reduction="sum") / n  # "sum", não "mean": ver o relatório
    kl = 0.5 * torch.sum(mu.pow(2) + logvar.exp() - 1.0 - logvar) / n
    return rec, kl


def kl_per_dim(mu, logvar):
    """KL média por imagem, separada por dimensão latente: um vetor (d,) cuja soma é a KL total."""
    return (0.5 * (mu.pow(2) + logvar.exp() - 1.0 - logvar)).mean(dim=0)


def sample_prior(model, n, generator=None):
    """Gera n imagens: z ~ N(0, I) do prior e x̂ = decoder(z)."""
    z = torch.randn(n, model.d, generator=generator)
    with torch.no_grad():
        return model.decode(z)


@torch.no_grad()
def evaluate(model, x, beta=1.0, batch_size=1000, seed=0):
    """Reconstrução e KL médias num conjunto inteiro, com o z amostrado como no treino.

    O ruído ε vem de um gerador próprio, com semente fixa. Assim a avaliação dá o mesmo número
    sempre e não mexe no gerador global, que embaralha os lotes do treino.
    """
    model.eval()
    g = torch.Generator().manual_seed(seed)
    rec_sum, kl_sum = 0.0, 0.0
    for i in range(0, len(x), batch_size):
        xb = x[i:i + batch_size]
        x_hat, mu, logvar = model(xb, generator=g)
        rec, kl = loss_terms(xb, x_hat, mu, logvar)
        rec_sum += rec.item() * len(xb)
        kl_sum += kl.item() * len(xb)
    model.train()
    rec, kl = rec_sum / len(x), kl_sum / len(x)
    return {"rec": rec, "kl": kl, "neg_elbo": rec + beta * kl}


def train(d, x_train, x_val, beta=1.0, lr=1e-3, batch_size=128, epochs=20, seed=42):
    """Item 1C: Adam, η = 10⁻³, lotes de 128, 20 épocas.

    A cada época, grava as duas parcelas separadas: a média da época no treino e o valor no
    fim da época na validação. O teste não entra aqui.
    """
    torch.manual_seed(seed)  # antes de construir o modelo, como pede o enunciado
    model = VAE(d)
    opt = torch.optim.Adam(model.parameters(), lr=lr)
    history = {"train_rec": [], "train_kl": [], "val_rec": [], "val_kl": []}

    for epoch in range(1, epochs + 1):
        perm = torch.randperm(len(x_train))  # nova ordem a cada época (gerador global, semeado acima)
        rec_sum, kl_sum = 0.0, 0.0
        for i in range(0, len(x_train), batch_size):
            xb = x_train[perm[i:i + batch_size]]
            x_hat, mu, logvar = model(xb)
            rec, kl = loss_terms(xb, x_hat, mu, logvar)
            loss = rec + beta * kl  # a perda é a média no lote do −ELBO por imagem
            opt.zero_grad()
            loss.backward()
            opt.step()
            rec_sum += rec.item() * len(xb)  # pondera pelo tamanho: o último lote é menor
            kl_sum += kl.item() * len(xb)

        val = evaluate(model, x_val, beta)
        history["train_rec"].append(rec_sum / len(x_train))
        history["train_kl"].append(kl_sum / len(x_train))
        history["val_rec"].append(val["rec"])
        history["val_kl"].append(val["kl"])
        print(f"d={d} época {epoch:2d}: treino rec {history['train_rec'][-1]:7.2f} "
              f"KL {history['train_kl'][-1]:6.2f} | val rec {val['rec']:7.2f} KL {val['kl']:6.2f}")

    model.eval()
    return model, history
    Parte Camadas
    Encoder \(784 \to 512 \to 256\), ReLU depois de cada uma
    Cabeças duas lineares \(256 \to d\): uma para \(\boldsymbol{\mu}\), outra para \(\log \boldsymbol{\sigma}^2\)
    Decoder \(d \to 256 \to 512 \to 784\), ReLU nas ocultas, sigmoide na saída

    Reparametrização (reparameterize). Sortear \(\mathbf{z} \sim \mathcal{N}(\boldsymbol{\mu}, \operatorname{diag} \boldsymbol{\sigma}^2)\) diretamente não deixa o gradiente passar pelo sorteio. Então o sorteio vai para um ruído que não depende dos parâmetros:

    \[ \mathbf{z} = \boldsymbol{\mu} + \boldsymbol{\sigma} \odot \boldsymbol{\varepsilon}, \qquad \boldsymbol{\varepsilon} \sim \mathcal{N}(\mathbf{0}, \mathbf{I}), \qquad \boldsymbol{\sigma} = \exp\left(\tfrac{1}{2} \log \boldsymbol{\sigma}^2\right) \]

    Agora \(\mathbf{z}\) é uma função diferenciável de \(\boldsymbol{\mu}\) e de \(\boldsymbol{\sigma}\), e o gradiente chega ao encoder.

    Perda (loss_terms), por imagem, com \(\beta = 1\) e média no lote:

    \[ L = \underbrace{-\sum_{j=1}^{784} \left[ x_j \log \hat{x}_j + (1 - x_j) \log (1 - \hat{x}_j) \right]}_{\text{reconstrução (BCE)}} + \beta \cdot \underbrace{\frac{1}{2} \sum_{i=1}^{d} \left( \mu_i^2 + \sigma_i^2 - 1 - \log \sigma_i^2 \right)}_{\text{KL}\left(q(\mathbf{z} \mid \mathbf{x}) \,\|\, \mathcal{N}(\mathbf{0}, \mathbf{I})\right)} \]

    A KL está escrita em forma fechada: 0.5 * torch.sum(mu.pow(2) + logvar.exp() - 1.0 - logvar) / n. A BCE é \(-\log p(\mathbf{x} \mid \mathbf{z})\) sob uma Bernoulli por pixel. A saída sigmoide é a probabilidade de cada pixel estar aceso.

    C — Train

    train usa Adam com \(\eta = 10^{-3}\), lotes de 128 e 20 épocas. A cada época, grava as duas parcelas separadas: no treino, a média da época, ponderada pelo tamanho de cada lote; na validação, o valor no fim da época.

    Conjunto Reconstrução KL \(-\)ELBO
    Validação, época 1 129.70 12.48 142.18
    Validação, época 20 82.86 19.35 102.21
    Teste 82.16 19.37 101.53

    Todos os valores estão em nats por imagem. O teste fica perto da referência do enunciado (cerca de 81 de reconstrução e 20 de KL).

    D — Reconstructions and samples

    E — Analysis

    Por que a KL sobe enquanto a perda total cai (Figura 1). A KL mede quanta informação sobre \(\mathbf{x}\) o código \(\mathbf{z}\) leva. Com KL zero, \(q(\mathbf{z} \mid \mathbf{x})\) é igual ao prior para toda imagem. Então \(\mathbf{z}\) não diz nada sobre \(\mathbf{x}\), e o decoder só pode devolver o "dígito médio". Essa é uma reconstrução muito cara. Cada nat de KL que o encoder gasta compra um código mais específico, e o decoder reconstrói melhor. No começo do treino, a troca é muito vantajosa: da época 1 para a 2, a KL de validação sobe 3.26 nats e a reconstrução cai 22.77 nats. O otimizador minimiza a soma, não cada parcela. Por isso ele aceita pagar mais KL sempre que a reconstrução cai mais do que isso. Ao longo das 20 épocas, a KL de validação vai de 12.48 para 19.35 nats (+6.87), e a reconstrução cai de 129.70 para 82.86 (−46.84). A troca para quando um nat a mais de código já não compra um nat a mais de reconstrução: depois da época 10, a KL quase não se move.

    O mesmo cálculo aparece em cada dimensão. Das 16 dimensões, só 11 têm KL acima de 0.1 nat no teste (de 0.85 a 2.55 nats cada). As outras 5 têm KL de 0.001 nat ou menos: nelas o encoder devolve o próprio prior, e o decoder ignora a coordenada. O otimizador só paga KL onde ela compra reconstrução.

    Por que a Figura 3 é pior do que a Figura 2. São dois motivos diferentes.

    1. Onde as amostras do prior caem. O encoder não preenche o prior por igual. Os códigos das imagens reais formam grupos, com regiões quase vazias entre eles. Em média, a dispersão total dos códigos é a do prior (variância agregada de 1.000 por dimensão), mas a forma não é. Por isso, uma parte dos \(\mathbf{z}\) sorteados do prior cai nesses buracos. Medi isso nas 11 dimensões ativas (nas outras 5, o decoder ignora a coordenada). A distância mediana de uma amostra do prior até o \(\boldsymbol{\mu}\) de teste mais próximo é 1.68, contra 1.29 para o \(\boldsymbol{\mu}\) de uma imagem de validação. Uma imagem real de validação fica a mais de uma certa distância dos códigos de teste em só 5% dos casos; uma amostra do prior, em 32.9% dos casos. Então cerca de um terço das amostras cai numa região onde quase não há dados: o decoder quase nunca recebeu um \(\mathbf{z}\) desses no treino. Então ele devolve uma mistura de formas, que às vezes não é dígito nenhum (na Figura 3: linha 1, coluna 3; linha 2, coluna 2; linha 5, coluna 1; linha 8, coluna 3).
    2. O que o decoder devolve para um \(\mathbf{z}\). A saída \(\hat{\mathbf{x}}\) não é uma imagem sorteada. É a média de \(p(\mathbf{x} \mid \mathbf{z})\): a probabilidade de cada pixel estar aceso. A BCE é mínima quando \(\hat{x}_j\) é a média dos pixels de todas as imagens compatíveis com aquele \(\mathbf{z}\). Quando várias formas são compatíveis (traço mais à esquerda ou mais à direita, mais grosso ou mais fino), a média é um borrão cinza. Medi a fração de pixels com valor entre 0.1 e 0.9: 8.6% nas imagens de teste, 15.0% nas reconstruções de \(\boldsymbol{\mu}\) e 16.8% nas amostras do prior. As reconstruções já são borradas pelo mesmo motivo. As amostras são ainda mais, porque um \(\mathbf{z}\) longe dos dados é compatível com mais formas.

    Por que decodificar \(\boldsymbol{\mu}\) na Figura 2. \(\boldsymbol{\mu}\) é o código mais provável de \(q(\mathbf{z} \mid \mathbf{x})\): o centro da região que o encoder reserva para aquela imagem. Um \(\mathbf{z}\) sorteado soma ao centro um ruído \(\boldsymbol{\sigma} \odot \boldsymbol{\varepsilon}\). Esse ruído leva o código para os vizinhos, e a figura mostraria uma variação aleatória da imagem, diferente a cada execução. A figura quer mostrar o que o modelo guardou da imagem, não o ruído do sorteio. No teste, decodificar \(\boldsymbol{\mu}\) dá 77.17 nats de reconstrução, contra 82.16 com \(\mathbf{z}\) sorteado.

Treina o mesmo modelo do Exercício 1 com d = 2 e produz as Figuras 4, 5 e 6. A linha de baixo
da Figura 6 usa o modelo d = 16 do Exercício 1. Também mede os números do item 2E.

import matplotlib.pyplot as plt
import numpy as np
import torch
from scipy.stats import norm

from common import save, tile
from vae import evaluate, train

D = 2
GRID = 15
Q = np.linspace(0.05, 0.95, GRID)  # quantis igualmente espaçados
Z_GRID = norm.ppf(Q)               # z = Φ⁻¹(q): cada célula tem a mesma massa do prior
INTERP_DIGITS = (0, 1)             # a Figura 6 vai da primeira imagem de teste de um 0 até a de um 1
STEPS = 10


def figure4(mu, y):
    """Figura 4: μ das 10 000 imagens de teste, coloridas pelo dígito."""
    cmap = plt.get_cmap("tab10")
    fig, ax = plt.subplots(figsize=(8.5, 7.5), layout="constrained")
    for k in range(10):
        pts = mu[y == k]
        ax.scatter(pts[:, 0], pts[:, 1], s=3, alpha=0.5, color=cmap(k), label=str(k), rasterized=True)
    ax.set_xlabel("$\\mu_1$")
    ax.set_ylabel("$\\mu_2$")
    ax.set_title("Figura 4 — μ das 10 000 imagens de teste no espaço latente (d = 2)")
    ax.legend(title="Dígito", markerscale=5, loc="upper left", bbox_to_anchor=(1.01, 1))
    ax.grid(alpha=0.3)
    save(fig, "fig4_latent_scatter.png")


def figure5(model):
    """Figura 5: decodifica a grade 15 × 15 em z = Φ⁻¹(q) e monta as imagens na mesma disposição.

    A coluna é z₁, crescendo da esquerda para a direita; a linha é z₂, crescendo de baixo para
    cima. Assim a Figura 5 tem a mesma orientação da Figura 4.
    """
    z1, z2 = np.meshgrid(Z_GRID, Z_GRID[::-1])  # linha 0 (topo) = maior z₂
    z = torch.tensor(np.column_stack([z1.ravel(), z2.ravel()]), dtype=torch.float32)
    with torch.no_grad():
        imgs = model.decode(z)
    fig, ax = plt.subplots(figsize=(9, 9.3), layout="constrained")
    ax.imshow(tile(imgs, GRID, GRID), cmap="gray", vmin=0, vmax=1)
    centers = [28 * c + 13.5 for c in range(GRID)]
    ax.set_xticks(centers, [f"{v:.2f}" for v in Z_GRID], rotation=90, fontsize=8)
    ax.set_yticks(centers, [f"{v:.2f}" for v in Z_GRID[::-1]], fontsize=8)
    ax.set_xlabel("$z_1 = \\Phi^{-1}(q)$, q de 0.05 a 0.95")
    ax.set_ylabel("$z_2 = \\Phi^{-1}(q)$, q de 0.05 a 0.95")
    ax.set_title("Figura 5 — O plano latente decodificado, grade 15 × 15 nos quantis do prior (d = 2)")
    save(fig, "fig5_latent_grid.png")


def interpolation(model16, x_test, y_test):
    """As duas linhas da Figura 6: no espaço dos pixels e no espaço latente do modelo d = 16."""
    a, b = (int((y_test == k).nonzero()[0]) for k in INTERP_DIGITS)
    t = torch.linspace(0, 1, STEPS).unsqueeze(1)
    xa, xb = x_test[a], x_test[b]
    pixel_row = (1 - t) * xa + t * xb
    with torch.no_grad():
        mu, _ = model16.encode(torch.stack([xa, xb]))
        latent_row = model16.decode((1 - t) * mu[0] + t * mu[1])
    return a, b, t.squeeze(1), pixel_row, latent_row


def figure6(a, b, t, pixel_row, latent_row):
    """Figura 6: duas linhas de 10 imagens, t de 0 a 1."""
    fig, axes = plt.subplots(2, STEPS, figsize=(12, 3.1), layout="constrained")
    for k in range(STEPS):
        for row, imgs in enumerate([pixel_row, latent_row]):
            ax = axes[row, k]
            ax.imshow(imgs[k].view(28, 28), cmap="gray", vmin=0, vmax=1)
            ax.set_xticks([])
            ax.set_yticks([])
        axes[1, k].set_xlabel(f"t = {t[k]:.2f}", fontsize=9)
    axes[0, 0].set_ylabel("Pixels")
    axes[1, 0].set_ylabel("Latente\n(d = 16)")
    fig.suptitle(f"Figura 6 — Interpolação entre as imagens de teste #{a} (dígito {INTERP_DIGITS[0]}) "
                 f"e #{b} (dígito {INTERP_DIGITS[1]})")
    save(fig, "fig6_interpolation.png")


@torch.no_grad()
def nearest_distance(x, ref, chunk=2000):
    """Distância euclidiana de cada linha de x à linha mais próxima de ref."""
    return torch.cat([torch.cdist(x[i:i + chunk], ref).min(dim=1).values
                      for i in range(0, len(x), chunk)])


@torch.no_grad()
def knn_confusion(mu_ref, y_ref, mu, y, k=15):
    """Matriz de confusão (10, 10) de um k-NN em μ: rótulos de referência da validação, alvo no teste.

    Mede quanto os dígitos se sobrepõem no espaço latente: onde os vizinhos de um ponto são de
    outro dígito, as nuvens se misturam.
    """
    pred = []
    for i in range(0, len(mu), 1000):
        idx = torch.cdist(mu[i:i + 1000], mu_ref).topk(k, largest=False).indices
        pred.append(torch.mode(y_ref[idx], dim=1).values)
    pred = torch.cat(pred)
    conf = torch.zeros(10, 10, dtype=torch.long)
    for true, p in zip(y.tolist(), pred.tolist()):
        conf[true, p] += 1
    return conf


def top_confused_pairs(conf, n=5):
    """Os n pares de dígitos {i, j} com mais confusões, somando os dois sentidos."""
    pairs = [((i, j), int(conf[i, j] + conf[j, i])) for i in range(10) for j in range(i + 1, 10)]
    return sorted(pairs, key=lambda p: -p[1])[:n]


@torch.no_grad()
def analysis_numbers(model2, model16, data, pixel_row, latent_row):
    """Os números do item 2E."""
    (x_train, _), (x_val, y_val), (x_test, y_test) = data["train"], data["val"], data["test"]
    results = {}
    for name, model in [("d2", model2), ("d16", model16)]:
        mu_val, _ = model.encode(x_val)
        mu_test, _ = model.encode(x_test)
        conf = knn_confusion(mu_val, y_val, mu_test, y_test)
        results[f"knn15_accuracy_{name}"] = conf.diag().sum().item() / len(y_test)
        results[f"top_confused_pairs_{name}"] = [[list(p), c] for p, c in top_confused_pairs(conf)]

    mu2, _ = model2.encode(x_test)
    # Mediana de μ de cada dígito, e onde ela cai na grade da Figura 5 (quantil q = Φ(z)).
    med = {k: mu2[y_test == k].median(dim=0).values.tolist() for k in range(10)}
    results["median_mu_d2"] = {k: [round(v, 3) for v in m] for k, m in med.items()}
    results["median_mu_d2_quantile"] = {k: [round(float(norm.cdf(v)), 3) for v in m]
                                        for k, m in med.items()}

    # Quão longe cada quadro da Figura 6 está da imagem de treino mais próxima (pixels).
    results["fig6_nn_train_dist_pixel_row"] = [round(v, 2) for v in nearest_distance(pixel_row, x_train).tolist()]
    results["fig6_nn_train_dist_latent_row"] = [round(v, 2) for v in nearest_distance(latent_row, x_train).tolist()]
    gray = lambda r: [round(((r[k] > 0.1) & (r[k] < 0.9)).float().mean().item(), 3) for k in range(len(r))]
    results["fig6_gray_fraction_pixel_row"] = gray(pixel_row)
    results["fig6_gray_fraction_latent_row"] = gray(latent_row)
    return results


def run(data, model16, results16):
    (x_train, _), (x_val, _), (x_test, y_test) = data["train"], data["val"], data["test"]
    model2, history = train(D, x_train, x_val)  # mesmos dados, semente, β e orçamento; só d muda

    test = evaluate(model2, x_test)
    with torch.no_grad():
        mu2, _ = model2.encode(x_test)
    figure4(mu2.numpy(), y_test.numpy())
    figure5(model2)
    a, b, t, pixel_row, latent_row = interpolation(model16, x_test, y_test)
    figure6(a, b, t, pixel_row, latent_row)

    return {
        "d": D,
        "test": test,
        "test_d16": results16["test"],
        "history": history,
        "interpolation_test_indices": [a, b],
        "interpolation_digits": list(INTERP_DIGITS),
        "analysis": analysis_numbers(model2, model16, data, pixel_row, latent_row),
    }

    C — The decoded plane

    Cada eixo usa \(z = \Phi^{-1}(q)\), com \(q\) em 15 valores igualmente espaçados em \([0.05, 0.95]\) (scipy.stats.norm.ppf). A coluna é \(z_1\), crescendo para a direita; a linha é \(z_2\), crescendo para cima, na mesma orientação da Figura 4.

    Por que esse espaçamento. Como \(\Phi\) é a função de distribuição acumulada de \(\mathcal{N}(0, 1)\), pontos igualmente espaçados em \(q\) dividem cada eixo em faixas de mesma probabilidade sob o prior; e como as duas coordenadas são independentes no prior, cada célula da grade tem a mesma massa de probabilidade. Por isso, a grade fica densa perto da origem, onde o prior tem mais massa, e esparsa nas bordas.

    E — Analysis

    Quais dígitos se sobrepõem (Figura 4) e a mesma transição na Figura 5. Para medir a sobreposição, classifiquei cada \(\boldsymbol{\mu}\) de teste pelo voto dos 15 \(\boldsymbol{\mu}\) de validação mais próximos. Em \(d = 2\), esse classificador acerta 79.47%; em \(d = 16\), 94.84%. Os pares mais confundidos em \(d = 2\) são 4 ↔ 9 (566 confusões), 3 ↔ 5 (358), 3 ↔ 8 (174), 5 ↔ 8 (161) e 7 ↔ 9 (132). Na Figura 4, as nuvens roxa (4) e ciano (9) quase coincidem, à direita do centro: as medianas são \((0.33,\ 0.54)\) para o 4 e \((0.48,\ 0.47)\) para o 9. Logo abaixo do centro, o 8, o 5 e o 3 formam uma faixa vertical contínua: medianas em \((0.12,\ -0.46)\), \((-0.20,\ -0.90)\) e \((-0.13,\ -1.35)\).

    Na Figura 5, a transição 4 ↔ 9 está nas linhas \(z_2\) de 0.50 a 0.92. Em cada uma, indo para a direita, um 4 de topo aberto vira um 9 de laço fechado. A fronteira é inclinada: na linha \(z_2 = 0.70\), a troca acontece em \(z_1 \approx 0.33\); na linha \(z_2 = 0.92\), já em \(z_1 \approx -0.16\). Nas células da fronteira, as imagens são um 4 cujos dois braços de cima se encostam e começam a fechar o laço: o mesmo traço serve de 4 e de 9. A transição 3 ↔ 5 aparece na última linha (\(z_2 = -1.64\)): o 5, com a barra reta em cima, ocupa as colunas de \(-1.64\) a \(-0.50\); o 3, com a curva em cima, começa em \(z_1 = -0.33\). A célula da fronteira tem a barriga de baixo, comum aos dois dígitos, e um topo que é meio barra, meio curva. Um passo para cima (\(z_2 = -0.70\) e \(-0.92\)), aparecem 8 entre eles: é a faixa contínua 3–5–8 da Figura 4.

    O que \(d = 2\) joga fora. Dois números por imagem mal bastam para dizer qual é o dígito. Não sobra espaço para o estilo: a espessura do traço, a inclinação, o tamanho, os detalhes de cada escrita. A KL de \(d = 2\) é só 6.63 nats por imagem, contra 19.37 em \(d = 16\): o código leva menos de um terço da informação. O preço é 53.93 nats a mais de reconstrução. A Figura 5 mostra isso. Cada região do plano tem um único dígito "protótipo", liso e de espessura média. Dentro de uma região, o estilo muda pouco e devagar, sempre na mesma direção: os 1 da direita só giram, de inclinados (embaixo) para retos (em cima); os 7 do canto de cima são quase iguais entre si; os 0 da esquerda só mudam de largura. Não há um 7 com traço no meio, um 1 com base, um 4 fechado ou um 2 com laço, que existem no MNIST. Ao reconstruir uma imagem, o modelo devolve o protótipo mais próximo, não a escrita daquela pessoa. Às vezes nem a classe sobrevive: 20.53% dos códigos de teste ficam mais perto de outro dígito.

    As duas interpolações (Figura 6). A linha de cima não passa por dígitos. A interpolação nos pixels só troca o brilho: em cada quadro há um 0 com intensidade \(1 - t\) e um 1 com intensidade \(t\), um sobre o outro. No meio, a imagem é um "ø": dois objetos meio apagados. Entre \(t = 0.11\) e \(t = 0.78\), de 22% a 25% dos pixels ficam cinzentos (entre 0.1 e 0.9), contra 8.6% numa imagem real. O conjunto das imagens de dígitos não é convexo: a média de dois dígitos quase nunca é um dígito, e a reta entre eles sai desse conjunto.

    A linha de baixo passa por imagens com cara de dígito. Cada quadro sai do decoder, e o decoder aprendeu a devolver uma única forma escrita para cada código. Então a reta no espaço latente vira uma curva sobre o conjunto dos dígitos: o 0 se estreita, passa por formas parecidas com um 8 e um 2 (\(t = 0.44\) e \(0.56\)) e termina num 1. Esses dígitos do meio aparecem porque, no espaço latente, o caminho entre o 0 e o 1 atravessa as regiões de outros dígitos. Os quadros do meio são os mais borrados da linha: a reta passa por uma região com menos dados, o mesmo efeito do item 1E. Mesmo assim, cada quadro é um objeto só, com um traço só.
